# 视力中心蓝牙数据汇聚系统实现计划

**目标**: 构建一个完整的视力中心蓝牙数据汇聚系统，支持多设备数据接入、统一格式转换、标准化打印输出。

**架构**: 严格分层设计（API层 → 业务逻辑层 → 设备抽象层 → 硬件抽象层），模块化实现，避免代码重复。

**技术栈**: Python 3.14+, FastAPI (HTTP服务), SQLite (数据存储), 平台适配蓝牙库, Vue 3 + Element Plus (Web前端)

`web/src/stores/index.ts` 内容:
```typescript
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import axios from 'axios'

export interface CheckRecord {
  id: string
  metadata: {
    checkTime: string
    deviceType: string
    deviceId: string
  }
  patientInfo: {
    patientName: string
    patientId: string
  }
}

export const useAppStore = defineStore('app', () => {
  const records = ref<CheckRecord[]>([])
  const loading = ref(false)
  const apiClient = axios.create({
    baseURL: '/api'
  })

  const fetchRecords = async () => {
    loading.value = true
    try {
      const response = await apiClient.get('/records')
      if (response.data.code === 0) {
        records.value = response.data.data.records
      }
    } catch (error) {
      console.error('Failed to fetch records:', error)
    } finally {
      loading.value = false
    }
  }

  const printRecords = async (recordIds: string[]) => {
    try {
      const response = await apiClient.post('/print', { record_ids: recordIds })
      return response.data
    } catch (error) {
      console.error('Failed to print records:', error)
      throw error
    }
  }

  return {
    records,
    loading,
    fetchRecords,
    printRecords
  }
})
```

<script setup lang="ts">
import { ref, computed } from 'vue'
import { useAppStore, type CheckRecord } from '@/stores'
import { ElMessage, ElMessageBox } from 'element-plus'

const store = useAppStore()
const previewVisible = ref(false)
const previewRecord = ref<CheckRecord | null>(null)
const selectedRecords = ref<CheckRecord[]>([])

const printQueue = computed(() => store.records)

const handleSelectionChange = (selection: CheckRecord[]) => {
  selectedRecords.value = selection
}

const openPreview = (record: CheckRecord) => {
  previewRecord.value = record
  previewVisible.value = true
}

const printSingle = (record: CheckRecord) => {
  store.printRecords([record.id])
  ElMessage.success('打印任务已发送')
}

const printAll = async () => {
  if (printQueue.value.length === 0) {
    ElMessage.warning('队列为空')
    return
  }
  await ElMessageBox.confirm(`确定打印全部 ${printQueue.value.length} 条记录?`, '确认')
  const ids = printQueue.value.map(r => r.id)
  await store.printRecords(ids)
  ElMessage.success('全部打印任务已发送')
}

const clearQueue = async () => {
  await ElMessageBox.confirm('确定清空打印队列?', '确认')
  ElMessage.success('队列已清空')
}

const removeFromQueue = (record: CheckRecord) => {
  ElMessage.success(`已移除 ${record.patientInfo.patientName} 的记录`)
}

const printPreviewed = () => {
  if (previewRecord.value) {
    printSingle(previewRecord.value)
    previewVisible.value = false
  }
}

const formatTime = (timeStr: string) => {
  return new Date(timeStr).toLocaleString('zh-CN')
}
</script>

<template>
  <div class="history">
    <el-card shadow="hover">
      <template #header>
        <div class="card-header">
          <span>打印历史</span>
          <div>
            <el-date-picker

            />

              <el-icon><Search /></el-icon>
              搜索
            </el-button>
          </div>
        </div>
      </template>

      <el-table :data="filteredRecords" style="width: 100%">
        <el-table-column prop="id" label="记录ID" width="180" />
        <el-table-column prop="patientInfo.patientName" label="姓名" width="120" />
        <el-table-column prop="patientInfo.patientId" label="ID" width="120" />
        <el-table-column prop="metadata.deviceType" label="设备类型" width="120" />
        <el-table-column prop="metadata.checkTime" label="检查时间" />
        <el-table-column label="操作" width="120">
          <template #default="{ row }">
            <el-button size="small" type="primary" @click="reprint(row)">补打</el-button>
          </template>
        </el-table-column>
      </el-table>

      <el-pagination
        v-model:current-page="currentPage"
        v-model:page-size="pageSize"
        :total="totalRecords"
        :page-sizes="[10, 20, 50, 100]"
        layout="total, sizes, prev, pager, next, jumper"
        style="margin-top: 20px; justify-content: flex-end;"
      />
    </el-card>
  </div>
</template>

<script setup lang="ts">
import { ref, computed } from 'vue'
import { useAppStore, type CheckRecord } from '@/stores'
import { ElMessage } from 'element-plus'

const store = useAppStore()
const dateRange = ref<[Date, Date] | null>(null)
const searchKeyword = ref('')
const currentPage = ref(1)
const pageSize = ref(20)

const filteredRecords = computed(() => {
  let result = [...store.records]
  
  if (searchKeyword.value) {
    const keyword = searchKeyword.value.toLowerCase()
    result = result.filter(r => 
      r.patientInfo.patientName.toLowerCase().includes(keyword) ||
      r.patientInfo.patientId.toLowerCase().includes(keyword)
    )
  }
  
  return result.reverse()
})

const totalRecords = computed(() => filteredRecords.value.length)

const search = () => {
  ElMessage.success('搜索完成')
}

const reprint = (record: CheckRecord) => {
  store.printRecords([record.id])
  ElMessage.success('补打任务已发送')
}
</script>
